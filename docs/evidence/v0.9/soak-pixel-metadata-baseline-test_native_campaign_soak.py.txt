def png_chunk(kind, payload):
    return len(payload).to_bytes(4,"big") + kind + payload + zlib.crc32(kind + payload).to_bytes(4,"big")


# A plain synthetic image file exercises actual retained-file header/hash checks.
# It is never gameplay or screenshot evidence.
PNG_1080 = (b"\x89PNG\r\n\x1a\n" + png_chunk(b"IHDR", (1920).to_bytes(4,"big") + (1080).to_bytes(4,"big") + b"\x08\x02\x00\x00\x00")
            + png_chunk(b"IDAT", zlib.compress((b"\x00" + bytes(1920*3))*1080)) + png_chunk(b"IEND", b""))


def encoded(value):
    if isinstance(value, dict):
        if value.get("$type") == "Vector2": return value
        return {"$type": "Dictionary", "entries": [[key, encoded(item)] for key, item in value.items()]}
    if isinstance(value, list): return [encoded(item) for item in value]
    return value


def save(mission, payload):

    if mission != "canyon": payload["mission_id"] = mission
    return {"version": 1, "mission": launcher.MISSION_TAGS[mission], "payload": encoded(payload)}

class SoakProofTests(unittest.TestCase):
    def test_all_starting_missions_produce_seven_strict_actual_file_cycles(self):
        for mission in launcher.MISSION_TAGS:
            with self.subTest(mission=mission), tempfile.TemporaryDirectory() as temporary:
                proof = SoakProof(Path(temporary)/"output", mission)
                self.assertEqual(proof.validate(), [])
                self.assertEqual(len(proof.report["cycles"]), 7)
                self.assertEqual(proof.report["planned_missions"][0], mission)

    def test_both_native_and_independent_actual_hour_are_required(self):
        for native, independent in ((3599.9,3859.0),(3858.0,3599.9)):
            with self.subTest(native=native, independent=independent), tempfile.TemporaryDirectory() as temporary:
                proof = SoakProof(Path(temporary)/"output")
                proof.report["wall_elapsed_seconds"] = native
                proof.process_wall = independent
                self.assertTrue(proof.validate())

    def test_partial_reordered_and_reused_cycles_cannot_claim_one_hour(self):
        mutations = [lambda p:p.report["cycles"].pop(),
                     lambda p:p.report["planned_missions"].reverse(),
                     lambda p:p.report["cycles"][2].update(report_path=p.report["cycles"][0]["report_path"]),
                     lambda p:p.cycle_reports[3]["main_loop"].update(start_main_tick_count=0,end_main_tick_count=32000),
                     lambda p:p.report.update(one_hour_soak_evidence=False),
                     lambda p:p.report["transitions"].pop()]
        for index, mutate in enumerate(mutations):
            with self.subTest(case=index), tempfile.TemporaryDirectory() as temporary:
                proof = SoakProof(Path(temporary)/"output")
                mutate(proof)
                # A malformed manifest may intentionally have no slot for flush.
                if len(proof.report["cycles"]) != len(proof.cycle_reports):
                    proof.cycle_reports = proof.cycle_reports[:len(proof.report["cycles"])]
                self.assertTrue(proof.validate())

class ModeAndLaunchTests(unittest.TestCase):
    def test_mode_specific_bounds_and_prepare_distinction(self):
        self.assertEqual(launcher.wall_time_limit(False,False,None),900)
        self.assertEqual(launcher.wall_time_limit(True,False,None),5400)
        for soak, bounds in ((False,(60,1800)),(True,(3600,7200))):
            for value in bounds: self.assertEqual(launcher.wall_time_limit(soak,False,value),value)
            for value in (bounds[0]-1,bounds[1]+1):
                with self.assertRaises(ValueError): launcher.wall_time_limit(soak,False,value)
        with self.assertRaises(ValueError): launcher.wall_time_limit(True,True,None)

    def test_soak_and_prepare_are_rejected_before_owning_or_launching_anything(self):
        with tempfile.TemporaryDirectory() as temporary:
            isolation = Path(temporary)/"untouched"
            args = ["run_native_campaign.py","--isolation-root",str(isolation),"--soak","--prepare-only"]
