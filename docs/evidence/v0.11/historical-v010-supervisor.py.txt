#!/usr/bin/env python3
"""Private detached supervisor; observes strict launcher's own wait() status."""
from datetime import datetime, timezone
import json
import os
from pathlib import Path
import subprocess
import sys
import time

PROJECT = Path(__file__).resolve().parents[1]
COMMIT = "b3f0033a8df1c51d2d43f625133e87c227c4c5eb"
BUILD = PROJECT / "work/native-soak-v010-unpacked/outpost-exodus-linux"
RUN = PROJECT / "work/native-soak-v010-full"
START = PROJECT / "work/v010-native-soak-start.json"
RESULT = PROJECT / "work/v010-native-soak-supervisor-result.json"
LOG = PROJECT / "work/v010-native-soak.log"

def ticks(pid):
    return int(Path(f"/proc/{pid}/stat").read_text().rsplit(")", 1)[1].split()[19])

def write(path, value):
    temporary = path.with_suffix(".tmp")
    temporary.write_text(json.dumps(value, indent=2) + "\n")
    temporary.replace(path)

assert subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=PROJECT, text=True).strip() == COMMIT
assert subprocess.check_output(["git", "status", "--porcelain"], cwd=PROJECT, text=True) == ""
assert not RUN.exists() and not START.exists() and not RESULT.exists() and not LOG.exists()
argv = [sys.executable, str(PROJECT / "tools/run_native_campaign.py"), "--build-dir", str(BUILD),
        "--isolation-root", str(RUN), "--mission", "canyon", "--soak", "--max-seconds", "5400"]
environment = os.environ.copy()
environment["DISPLAY"] = ":0"
started = time.monotonic()
with LOG.open("w") as output:
    child = subprocess.Popen(argv, cwd=PROJECT, env=environment, stdin=subprocess.DEVNULL,
                             stdout=output, stderr=subprocess.STDOUT)
    record = {"schema_version": 1, "source_commit": COMMIT, "build_dir": str(BUILD),
              "isolation_root": str(RUN), "argv": argv, "started_utc": datetime.now(timezone.utc).isoformat(),
              "supervisor_pid": os.getpid(), "supervisor_start_ticks": ticks(os.getpid()),
              "launcher_pid": child.pid, "launcher_start_ticks": ticks(child.pid),
              "launcher_log": str(LOG)}
    write(START, record)
    returncode = child.wait()
record.update({"launcher_returncode": returncode, "launcher_exit_observed_by": "supervisor subprocess.Popen.wait",
               "launcher_wall_elapsed_seconds": time.monotonic() - started,
               "finished_utc": datetime.now(timezone.utc).isoformat(),
               "scope": "This observes the strict launcher's exit; native child and full-route acceptance remain in verification.json. Supervisor's own eventual exit status is not inferred."})
write(RESULT, record)
raise SystemExit(returncode)
