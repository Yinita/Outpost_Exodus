extends SceneTree
func _initialize() -> void:
	print("CMDLINE ", OS.get_cmdline_args())
	print("USERARGS ", OS.get_cmdline_user_args())
	var names: Array[String] = []
	for method: Dictionary in Engine.get_method_list():
		var name: String = method.name
		if name.contains("fixed") or name.contains("time") or name.contains("frame") or name.contains("movie"): names.append(name)
	print("ENGINE_CLOCK_METHODS ", names)
	print("TIME_SCALE ", Engine.time_scale)
	var file: FileAccess = FileAccess.open("/proc/self/cmdline", FileAccess.READ)
	var raw: PackedByteArray = file.get_buffer(65536) if file != null else PackedByteArray()
	if file != null: file.close()
	var parts: Array[String] = []
	var part: PackedByteArray = PackedByteArray()
	for byte: int in raw:
		if byte == 0:
			parts.append(part.get_string_from_utf8())
			part.clear()
		else: part.append(byte)
	if not part.is_empty(): parts.append(part.get_string_from_utf8())
	print("ACTUAL_PROC_ARGV ", parts)
	quit()
